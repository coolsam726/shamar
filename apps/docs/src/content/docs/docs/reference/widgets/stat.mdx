```ts
import { Stat, html } from '@shamar/core'

Stat.make('Unique views', '192.1k')
  .description('32k increase')
  .descriptionIcon('arrow-trending-up')
  .descriptionColor('success')
  .icon('eye')
  .color('success')
  .chart([7, 2, 10, 3, 15, 4, 17])
  .chartColor('success')
  .url('/admin/analytics')
  .openUrlInNewTab()
  .extraAttributes({ 'wire:click': '…' })
  .columnSpan(2)

Stat.make('Missing', null).placeholder('—')

Stat.configureUsing((stat) => {
  stat.placeholder('-')
})

Stat.make('Rich', html('<em>1.2k</em>'))
```
