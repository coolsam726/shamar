```ts
import { StatsOverviewWidget, Stat, pageFilters, html } from '@shamar/core'

export class ProductStatsWidget extends StatsOverviewWidget {
  static override sort = 10
  static override columnSpan = 'full'
  static override heading = 'Analytics'
  static override description = 'An overview of some analytics.'
  /** Cards per row (Filament getColumns). null = auto from count. */
  static override columns = 4
  /** Filament defaults: lazy + poll every 5s. */
  static override isLazy = false
  static override pollingInterval: string | null = '5s'

  static override async stats(ctx) {
    const filters = pageFilters(ctx) // dashboard filtersForm values
    return [
      Stat.make('Total products', count)
        .description('32k increase')
        .descriptionIcon('arrow-trending-up')
        .icon('cube')
        .color('success')
        .chart([7, 2, 10, 3, 15, 4, 17])
        .url('/admin/products')
        .openUrlInNewTab()
        .extraAttributes({ class: 'cursor-pointer' })
        .columnSpan(1),
      Stat.make('Views', null).placeholder('—'),
      Stat.make('Note', html('<strong>HTML</strong> value')),
    ]
  }
}
```

Scaffold with:

```bash
node ace shamar:make-widget ProductStats --type=stats
```

### Dashboard filters

```ts
class AdminDashboard extends DashboardPage {
  static override filtersForm(form) {
    return form.schema([
      Select.make('period').options([
        { label: 'Today', value: 'today' },
        { label: 'MTD', value: 'mtd' },
      ]),
    ])
  }
  static override widgets() {
    return [ProductStatsWidget]
  }
}
```
