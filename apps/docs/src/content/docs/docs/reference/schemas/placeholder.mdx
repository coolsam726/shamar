import DocApiTable from '../../../../../components/DocApiTable.astro';

`Placeholder` is a labeled, field-like static slot (Filament Placeholder). For untitled prose, use [Text](/docs/reference/schemas/text/).

```ts
import { Placeholder, Section } from '@shamar/core'

Section.make('Computed').schema([
  Placeholder.make('full_name')
    .label('Full name')
    .content('Built from first + last on save.'),
])
```
