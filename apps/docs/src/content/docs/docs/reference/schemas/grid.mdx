import DocApiTable from '../../../../../components/DocApiTable.astro';

`Grid` lays children out in columns without wrapping them in a section card.

```ts
import { Grid, Section, TextInput } from '@shamar/core'

Section.make('Details').schema([
  Grid.make(3).schema([
    TextInput.make('width'),
    TextInput.make('height'),
    TextInput.make('depth'),
  ]),
])
```

Children can still call `.columnSpan()` / `.columnSpanFull()` to span multiple tracks.
