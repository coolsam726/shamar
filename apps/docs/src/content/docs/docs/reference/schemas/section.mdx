import DocApiTable from '../../../../../components/DocApiTable.astro';

`Section` is the primary Filament-style card for grouping fields or infolist entries.

```ts
import { Section, TextInput } from '@shamar/core'

Section.make('Catalog')
  .icon('cube')
  .description('Product identity and pricing.')
  .columns(2)
  .collapsible()
  .schema([
    TextInput.make('name').required(),
    TextInput.make('sku').maxLength(32),
  ])
```
