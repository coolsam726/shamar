import DocApiTable from '../../../../../components/DocApiTable.astro';

`Fieldset` mirrors Filament’s fieldset: a labeled boundary without the elevated card of [Section](/docs/reference/schemas/section/).

```ts
import { Fieldset, TextInput } from '@shamar/core'

Fieldset.make('Address').columns(2).schema([
  TextInput.make('city'),
  TextInput.make('country'),
])
```
